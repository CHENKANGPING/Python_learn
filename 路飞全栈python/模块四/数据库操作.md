# 数据库操作

```sql
-- 查看当前使用的数据库
select database();

-- 查看所有数据库
show databases();
-- 模糊查询数据库
show databases like "....";
-- 匹配查询数据库
show databases like "...%";

-- 创建数据库（在磁盘上创建一个对应的文件夹）
create database [if not exists] db_name [character set ...];
-- 查看数据库的创建方式
show create database db_name;

-- 修改数据库
alter database db_name [character set ...];

-- 删除数据库
drop database [if exists] db_name;

-- 使用数据库
use db_name; -- 切换数据库 注：进入到某个数据库后没办法再回到之前状态，但可以通过use进行切换
-- 数据库备份
mysqldump -u username -p password database_name > backup.sql

```
